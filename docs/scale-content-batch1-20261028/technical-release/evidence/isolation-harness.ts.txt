async function snapshot(name:string){
 const users=await db.user.findMany({select:{id:true,username:true,role:true,platformRole:true,accountDomain:true,isActive:true,isFrozen:true},orderBy:{id:'asc'}})

 for(const s of scales){s.definitionCanonicalHash=canonicalHash(s.definition);delete s.definition}
 const deployments=await db.$queryRawUnsafe('SELECT * FROM scale_deployment_policies ORDER BY id')
 const axes=await db.$queryRawUnsafe('SELECT * FROM scale_axis_snapshots ORDER BY scale_id,axis_hash')
 for(const a of axes){const s=a.source_snapshot;a.observedSourceCanonicalHash=canonicalHash(s);a.frozenIdentity=s.identity;a.frozenReleaseStatus=s.executable.releaseStatus;a.frozenInstruction=s.instruction;a.frozenManual={catalogStatus:s.catalog.catalogStatus,maturity:s.catalog.scientificMaturity,localization:s.localization,scientificReview:s.scientificReview??null,license:s.executable.definition.license};delete a.source_snapshot}
 const audits=await db.scalePublicationAudit.findMany({orderBy:{id:'asc'}})
 const releases=await db.measurementContentRelease.findMany({select:{id:true,layer:true,family:true,key:true,version:true,contentHash:true,scope:true,provenance:true},orderBy:{id:'asc'}})
 const policies=await db.$queryRawUnsafe('SELECT product,resource_key,fingerprint,decision,paused,reason,reviewed_by,version FROM survey_tool_policies ORDER BY product,resource_key,fingerprint')
 const counts:any={}
 for(const table of ['assessment_reference_sets','measurement_content_release_events','content_capability_grants','material_grants','survey_tool_grants','scale_library_entries','scale_library_use_grants','education_assessment_deliveries','assessments']){const rows=await db.$queryRawUnsafe('SELECT count(*)::int AS count FROM '+table);counts[table]=rows[0].count}
 const value=JSON.parse(JSON.stringify({users,scales,deployments,axes,audits,releases,policies,counts},(_k,v)=>typeof v==='bigint'?v.toString():v))

}
async function cli(label:string,script:string,args:string[],expected:number,readonly=false){
 const before=await snapshot(label+'-before'),start=new Date().toISOString()
 const argv=['node_modules/tsx/dist/cli.mjs','src/scripts/'+script,...args]
 const result=spawnSync(process.execPath,argv,{cwd:backend,env:{...process.env,NODE_ENV:'test'},encoding:'utf8',timeout:60000,maxBuffer:8*1024*1024})
 const logs:any={}
 for(const [kind,text] of [['stdout',result.stdout??''],['stderr',result.stderr??'']]){writeFileSync(out+'/'+label+'.'+kind+'.log',text);logs[kind]={path:label+'.'+kind+'.log',bytes:Buffer.byteLength(text),sha256:sha(text)}}
 const after=await snapshot(label+'-after')
 const row={label,cwd:backend,argv:[process.execPath,...argv],startedAt:start,endedAt:new Date().toISOString(),processExitCode:result.status,signal:result.signal,error:result.error?.message??null,expectedExitCode:expected,readOnly:readonly,beforeAfterIdentical:canonicalHash(before)===canonicalHash(after),logs}
 report.commands.push(row);save();check(label+' actual exit',result.status===expected,{actual:result.status,expected})
 if(readonly)check(label+' made no tracked business writes',row.beforeAfterIdentical)
 return {data:result.status===0?JSON.parse(result.stdout):null,before,after,row}
}

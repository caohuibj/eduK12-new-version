import {writeFileSync} from 'node:fs'
import {createSartSnapshot} from '/workspace/eduK12-new-version/server-version/backend/src/modules/cognitive/content-candidates/sart/package'
import {sartPlan,type SartRaw} from '/workspace/eduK12-new-version/server-version/backend/src/modules/cognitive/content-candidates/sart/protocol'
import {scoreSart,sartEnvelope} from '/workspace/eduK12-new-version/server-version/backend/src/modules/cognitive/content-candidates/sart/scorer'
import {projectSartReport,sartDownload,sartImageData} from '/workspace/eduK12-new-version/server-version/backend/src/modules/cognitive/content-candidates/sart/report'
const session=createSartSnapshot('experience','2026-10-27T00:00:00.000Z'),randomSeed='sart-browser-synthetic',config=session.config,raw:SartRaw[]=sartPlan(config,randomSeed).map(p=>({plan:p,onsetMs:100+1150*p.index,endedAtMs:1250+1150*p.index,status:'complete',stopReason:null,heldBefore:[],events:p.digit===3?[{kind:'mask',atMs:350+1150*p.index}]:[{kind:'down',atMs:420+1150*p.index,token:'p:1',input:'touch',repeat:false},{kind:'up',atMs:421+1150*p.index,token:'p:1',input:'touch',repeat:false},{kind:'mask',atMs:350+1150*p.index}].sort((a,b)=>a.atMs-b.atMs) as SartRaw['events'],ignoredRepeats:0,overflow:false}))
const reports=Object.fromEntries([['interpretable',raw],['limited',raw.slice(0,6)],['invalid',[]]].map(([name,rows])=>{const report=projectSartReport(scoreSart({session,config,randomSeed,trials:(rows as SartRaw[]).map(sartEnvelope)}));return [name,{report,download:JSON.parse(sartDownload(report)),image:sartImageData(report)}]}))
writeFileSync('/workspace/eduK12-new-version/docs/cognitive-library/scenario-roadmap-20261027/domains/clinical-research/implementation-evidence/preview-reports.json',JSON.stringify({kind:'AUTHORITATIVE_SOFTWARE_SYNTHETIC_LAYOUT_FIXTURE_NOT_PARTICIPANT_DATA',reports},null,2)+'\n')

import {describe,it,expect,vi} from 'vitest'
import type {Prisma} from '@prisma/client'
import {randomUUID,createHash} from 'node:crypto'
import {integrationDatabaseUrl,requireIsolatedReleaseDatabase} from '../integration/integration-env'
import {nativeCompletionFixture} from './native-completion-fixture.test-support'
import {nativeCognitiveConfigSeeds} from '../../modules/cognitive/content-candidates/native-registry/config-seeds'
import {decryptUnifiedRuntimePayload} from '../../modules/assessment-runtime/security'
import {canonicalHash} from '../../modules/assessment-runtime/canonical'
import {completionFromNativeClaim} from '../../modules/cognitive/native-completion'
vi.mock('dotenv',()=>({default:{config:()=>({parsed:{}})}}))
const url=integrationDatabaseUrl('COGNITIVE_INTEGRATION_DB_URL'),suite=url?describe:describe.skip
suite('native completion frozen in the committed FINAL',()=>{
 it('keeps all 32 closed HF windows, lowers the aborted report and binds the identical archived claim',async()=>{
  requireIsolatedReleaseDatabase(url!);Object.assign(process.env,{COGNITIVE_MODULE_ENABLED:'true',DATA_ENCRYPTION_KEY:'a'.repeat(64),DATA_PSEUDONYM_KEY:'b'.repeat(64)})
  const db=(await import('../../config/database')).prisma
  const assignment=await import('../../modules/cognitive/assignment.service'),sessions=await import('../../modules/cognitive/session.service'),release=await import('../../modules/cognitive/release.service'),final=await import('../../modules/cognitive/final-submit.service'),archive=await import('../../modules/cognitive/raw-archive.service'),security=await import('../../modules/cognitive/cognitive.security')
  const prefix='completion-'+randomUUID(),actors=[]
  try{
   for(const role of ['TEACHER','STUDENT','ADMIN'] as const)actors.push(await db.user.create({data:{username:prefix+'-'+role,passwordHash:'synthetic-no-login',role,platformRole:role==='ADMIN'?'SYSTEM_ADMIN':'STANDARD',isActive:true}}))
   const [teacher,student,publisher]=actors,course=await db.course.create({data:{title:prefix,courseCode:randomUUID(),creatorId:teacher.id}})
   await db.courseStudent.create({data:{courseId:course.id,studentId:student.id,status:'ACTIVE'}})
   const seed=nativeCognitiveConfigSeeds.find(s=>s.testType==='hearts-flowers')!
   const config=await db.cognitiveTestConfig.findUnique({where:{testType_configVersion:{testType:seed.testType,configVersion:seed.configVersion}}}) ?? await db.cognitiveTestConfig.create({data:{...seed,config:seed.config as Prisma.InputJsonValue}})
   await release.publishCognitiveConfig(config.id,publisher.id)
   const a=await assignment.createAssignment(teacher.id,'TEACHER',{courseId:course.id,configId:config.id,title:prefix,profile:'experience',required:true,maxAttempts:1})
   await assignment.publishAssignment(teacher.id,'TEACHER',a.id)
   const session=await sessions.createSession(student.id,a.id),f=nativeCompletionFixture(session.sessionId,session.attemptEpoch)
   await db.cognitiveSession.update({where:{id:session.sessionId},data:{randomSeed:f.input.randomSeed}})
   expect(session.sessionSnapshot!.configHash).toBe(f.input.session.configHash)
   const input={sessionId:session.sessionId,submissionId:prefix+'-final',attemptEpoch:session.attemptEpoch,definitionHash:f.input.session.configHash,contextSnapshotHash:session.contextSnapshotHash,trials:f.input.trials,nativeCompletion:f.completion}
   for(const changed of [{formalCount:31},{manifestHash:'f'.repeat(64)},{contextKey:'other-attempt'}]){
    await expect(final.submitCognitiveSessionFinal(student.id,{...input,nativeCompletion:{...f.completion,...changed}})).rejects.toMatchObject({code:'SUBMISSION_PAYLOAD_CONFLICT'})
   }
   const saved=await final.submitCognitiveSessionFinal(student.id,input),row=await db.cognitiveSession.findUniqueOrThrow({where:{id:session.sessionId}})
   const result=(await sessions.getSession(student.id,session.sessionId) as {result:any}).result
   expect(result.quality.state).toBe('limited');expect(result.metrics.validTrialCount).toBe(32);expect(result.metrics.receivedTrialCount).toBe(32)
   expect(result.report.nativeReport).toMatchObject({state:'limited',testType:'hearts-flowers'})
   const rawRow=await db.cognitiveRawSubmission.findUniqueOrThrow({where:{sessionId_attemptEpoch:{sessionId:session.sessionId,attemptEpoch:session.attemptEpoch}}})
   const core=decryptUnifiedRuntimePayload<any>(rawRow.payloadEncrypted),canonical={trials:f.input.trials,nativeCompletion:f.completion}
   expect(core.nativeCompletion).toEqual(f.completion);expect(core.trials).toEqual(f.input.trials);expect(saved.payloadHash).toBe(canonicalHash(canonical))
   expect(saved.payloadHash).not.toBe(canonicalHash({trials:f.input.trials}))
   const identity={sessionId:session.sessionId,attemptEpoch:session.attemptEpoch,submissionId:input.submissionId,definitionHash:input.definitionHash},deps={db,storeAsset:vi.fn(async(value:any)=>{
    const asset=await db.storedAsset.create({data:{id:randomUUID(),objectKey:'assets/'+randomUUID()+'.enc',provider:'cos',accessScope:'PRIVATE',ownerId:student.id,scopeId:value.scopeId,mimeType:value.mimeType,originalName:value.originalName,sizeBytes:value.buffer.length,sha256:createHash('sha256').update(value.buffer).digest('hex'),references:{create:value.initialReference}}})
    expect(security.decryptCognitivePayload<any>(value.buffer.toString()).detail.claim).toEqual(f.claim);return asset
   }),discardUnreferencedAsset:vi.fn()}
   const tampered=structuredClone(f.detail);tampered.claim.terminationReason='changed-after-FINAL'
   const {finalPayloadDigest:old,...body}=tampered.claim;tampered.claim.finalPayloadDigest=canonicalHash(body)
   expect(completionFromNativeClaim(tampered.claim)).not.toEqual(f.completion)
   await expect(archive.archiveCognitiveRawSubmission(identity,{userId:student.id},{rawPayload:tampered,rawSha256:canonicalHash(tampered),expectedCoreHash:saved.payloadHash},deps)).rejects.toMatchObject({code:'ARCHIVE_IDENTITY_CONFLICT'})
   expect(deps.storeAsset).not.toHaveBeenCalled()
   const uploaded=await archive.archiveCognitiveRawSubmission(identity,{userId:student.id},{rawPayload:f.detail,rawSha256:canonicalHash(f.detail),expectedCoreHash:saved.payloadHash},deps)
   expect(uploaded.archiveConfirmed).toBe(true);expect(deps.storeAsset).toHaveBeenCalledTimes(1)
   expect((await db.cognitiveSession.findUniqueOrThrow({where:{id:session.sessionId}})).resultSnapshotEncrypted).toBe(row.resultSnapshotEncrypted)
  }finally{await db.$disconnect()} // The whole dedicated DB is disposable; immutable audit releases are retained until removal.
 },60000)
})

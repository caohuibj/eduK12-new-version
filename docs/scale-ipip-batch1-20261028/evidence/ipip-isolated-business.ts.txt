 for(const a of axes){const s=a.source_snapshot;a.observedSourceCanonicalHash=canonicalHash(s);a.frozenIdentity=s.identity;a.frozenReleaseStatus=s.executable.releaseStatus;a.frozenInstruction=s.instruction;a.frozenManual={catalogStatus:s.catalog.catalogStatus,maturity:s.catalog.scientificMaturity,localization:s.localization,scientificReview:s.scientificReview??null,license:s.executable.definition.license};delete a.source_snapshot}

async function operation(label:string,api:string,input:any,call:()=>Promise<any>,readonly=false){
 const before=await snapshot(label+'-before');const start=new Date().toISOString()
 try{
 const result=await call();const after=await snapshot(label+'-after')
 const row={label,api,input,startedAt:start,endedAt:new Date().toISOString(),outcome:'RESOLVED',result,readOnly:readonly,beforeAfterIdentical:canonicalHash(before)===canonicalHash(after)}
 report.commands.push(row);save();if(readonly)check(label+' preview has no tracked writes',row.beforeAfterIdentical)
 return {result,before,after}
 }catch(e:any){report.commands.push({label,api,input,startedAt:start,endedAt:new Date().toISOString(),outcome:'REJECTED',name:e.name,message:e.message});save();throw e}
}

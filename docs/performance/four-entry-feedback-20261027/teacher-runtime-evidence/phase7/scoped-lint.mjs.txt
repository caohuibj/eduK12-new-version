import {ESLint} from '/workspace/eduK12-new-version/server-version/frontend/node_modules/eslint/lib/api.js'
import config from '/workspace/eduK12-new-version/server-version/frontend/eslint.config.js'
import globals from '/workspace/eduK12-new-version/server-version/frontend/node_modules/globals/index.js'
import fs from 'node:fs'
const cwd='/workspace/eduK12-new-version/server-version/backend'
const files=['src/__tests__/huiteacher/assessment-scale-marker-final.postgres.integration.test.ts','src/__tests__/huiteacher/assessment-scale-marker.postgres.fixture.ts']

const results=await lint.lintFiles(files),formatter=await lint.loadFormatter('stylish'),output=formatter.format(results)

const code=results.some(r=>r.errorCount||r.warningCount)?1:0

process.stdout.write(output);process.exitCode=code

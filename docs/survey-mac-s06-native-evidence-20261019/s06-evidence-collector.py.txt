"""Ephemeral loopback-only native-UI evidence receiver; no app API proxy."""
from http.server import ThreadingHTTPServer,BaseHTTPRequestHandler
from pathlib import Path
import json,base64,hashlib,re
OUT=Path(__file__).resolve().parents[1]/'evidence/s06/native'
HTML='''<!doctype html><meta charset="utf-8"><title>S06 本机证据接收</title><h1>S06 本机证据接收</h1><p>只保存本任务 CUA 原始截图或 AX 字节，不调用产品接口。</p><form><label>证据文件名<input id="name" autocomplete="off"></label><label>预期 FNV<input id="fnv" autocomplete="off"></label><label>Base64 字节<textarea id="data" style="display:block;width:90%;height:250px" spellcheck="false"></textarea></label><button>保存证据</button></form><output id="result" role="status"></output><script>document.querySelector('form').onsubmit=async e=>{e.preventDefault();document.querySelector('#result').textContent='正在校验';try{let r=await fetch('/',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({name:document.querySelector('#name').value,fnv:document.querySelector('#fnv').value,data:document.querySelector('#data').value})});document.querySelector('#result').textContent=await r.text();document.querySelector('#data').value=''}catch(e){document.querySelector('#result').textContent='接收失败 '+e.message}}</script>'''.encode()
class Handler(BaseHTTPRequestHandler):
 def log_message(self,*args):pass
 def do_GET(self):
  if self.path!='/':self.send_error(404);return
  self.send_response(200);self.send_header('Content-Type','text/html;charset=utf-8');self.send_header('Cache-Control','no-store');self.end_headers();self.wfile.write(HTML)
 def do_POST(self):
  try:
   assert self.path=='/' and self.headers.get('Origin')=='http://127.0.0.1:3198'
   length=int(self.headers['Content-Length']);assert 0<length<20*1024*1024
   value=json.loads(self.rfile.read(length));name=value['name']
   assert re.fullmatch(r'[a-z0-9-]+\.(?:jpg|json)',name)
   data=base64.b64decode(value['data'],validate=True);h=2166136261
   for n in data:h=((h^n)*16777619)&0xffffffff
   assert format(h,'08x')==value['fnv']
   if name.endswith('.jpg'):assert data.startswith(b'\xff\xd8\xff')
   else:json.loads(data)
   path=OUT/('screenshots/'+name if name.endswith('.jpg') else name)
   with path.open('xb') as f:f.write(data)
   receipt={'name':name,'bytes':len(data),'fnv1a32':format(h,'08x'),'sha256':hashlib.sha256(data).hexdigest(),'saved':True}
   with (OUT/'collector-receipts.jsonl').open('a') as f:f.write(json.dumps(receipt)+'\n')
   self.send_response(200);self.end_headers();self.wfile.write(json.dumps(receipt).encode())
  except Exception as e:
   self.send_response(400);self.end_headers();self.wfile.write(('未保存 '+type(e).__name__).encode())
ThreadingHTTPServer(('127.0.0.1',3198),Handler).serve_forever()
